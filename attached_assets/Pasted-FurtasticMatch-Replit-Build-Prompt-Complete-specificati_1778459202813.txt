function scoreBreed(breed, answers) {
  let score = 0;

  // Activity match (20% weight)
  const activityDiff = Math.abs(breed.activity_level * 10 - answers.activity);
  score += (1 - activityDiff / 100) * 20;

  // Space match (15% weight)
  const spaceScores = { apartment: 1, 'house-no-yard': 2, 'house-yard': 3, rural: 4 };
  const spaceDiff = Math.abs(breed.space_required - spaceScores[answers.space]);
  score += (1 - spaceDiff / 3) * 15;

  // Grooming match (15% weight)
  const groomingDiff = Math.abs(breed.grooming_needs * 10 - answers.grooming);
  score += (1 - groomingDiff / 100) * 15;

  // Experience match (15% weight)
  const expScores = { first: 1, some: 2, seasoned: 3 };
  const expDiff = Math.abs(breed.experience_required - expScores[answers.experience]);
  score += (1 - expDiff / 2) * 15;

  // Size match (10% weight)
  if (answers.size === 'any' || breed.size === answers.size) score += 10;

  // Kids score (15% weight)
  const kidsImportance = answers.kids / 100;
  score += breed.good_with_kids * kidsImportance * 15;

  // Who match (10% weight)
  if (answers.who === 'family' && breed.good_with_kids >= 4) score += 10;
  else if (answers.who === 'couple' || answers.who === 'solo') score += 5;

  return Math.round(score);
}
	3	Sort breeds by score descending
	4	If Q1 named breeds: boost those breeds by 15 points (ensures they appear in top 3)
	5	Top 3 = matches. Pick wildcard = 4th highest scoring breed NOT in top 3 that has wildcard_eligible: true
	6	Encode result as hash: btoa(JSON.stringify([breed1.id, breed2.id, breed3.id, wildcard.id]))
	7	Navigate to /results?r=[hash]
